cmake_minimum_required(VERSION 3.20)
project(lwIOlinkMasterTests LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(ARDUINOJSON_INCLUDE "" CACHE PATH "ArduinoJson 7.4.3 src directory")
if(NOT EXISTS "${ARDUINOJSON_INCLUDE}/ArduinoJson.h")
  message(FATAL_ERROR "Set ARDUINOJSON_INCLUDE to ArduinoJson 7.4.3/src")
endif()
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS src/vendor/iolinki_master/*.c src/vendor/iolinki/*.c)
add_library(iolink_core STATIC ${CORE_SOURCES} src/SinglePort.cpp src/UsbProtocol.cpp)
target_include_directories(iolink_core PUBLIC src PRIVATE "${ARDUINOJSON_INCLUDE}")
if(MSVC)
  target_compile_options(iolink_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(iolink_core PRIVATE -Wall -Wextra -fno-strict-aliasing)
endif()
enable_testing()
add_executable(integration_tests tests/integration.cpp third_party/iolinki-master/tests/fake_iolink_device.c)
target_include_directories(integration_tests PRIVATE third_party/iolinki-master/tests "${ARDUINOJSON_INCLUDE}")
target_link_libraries(integration_tests PRIVATE iolink_core)
if(MSVC)
  target_compile_options(integration_tests PRIVATE /utf-8)
endif()
add_test(NAME integration COMMAND integration_tests)
