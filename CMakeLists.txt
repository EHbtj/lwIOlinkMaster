cmake_minimum_required(VERSION 3.20)
project(lwIOlinkMasterTests LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(ARDUINOJSON_INCLUDE "" CACHE PATH "ArduinoJson 7.4.3 src directory")
if(NOT EXISTS "${ARDUINOJSON_INCLUDE}/ArduinoJson.h")
  message(FATAL_ERROR "Set ARDUINOJSON_INCLUDE to ArduinoJson 7.4.3/src")
endif()
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS src/vendor/iolinki_master/*.c src/vendor/iolinki/*.c)
add_library(iolink_core STATIC ${CORE_SOURCES} src/SinglePort.cpp src/UsbProtocol.cpp)
target_include_directories(iolink_core PUBLIC src PRIVATE "${ARDUINOJSON_INCLUDE}")
if(MSVC)
  target_compile_options(iolink_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(iolink_core PRIVATE -Wall -Wextra -fno-strict-aliasing)
endif()
enable_testing()
add_executable(integration_tests tests/integration.cpp third_party/iolinki-master/tests/fake_iolink_device.c)
target_include_directories(integration_tests PRIVATE third_party/iolinki-master/tests "${ARDUINOJSON_INCLUDE}")
target_link_libraries(integration_tests PRIVATE iolink_core)
if(MSVC)
  target_compile_options(integration_tests PRIVATE /utf-8)
endif()
add_test(NAME integration COMMAND integration_tests)
add_executable(startup_recovery_tests tests/startup_recovery.cpp
    third_party/iolinki-master/tests/fake_iolink_device.c)
target_include_directories(startup_recovery_tests PRIVATE third_party/iolinki-master/tests)
target_link_libraries(startup_recovery_tests PRIVATE iolink_core)
if(MSVC)
  target_compile_options(startup_recovery_tests PRIVATE /W4 /utf-8)
endif()
foreach(scenario timing ident_retries scan_errors transient_errors repeated_errors partial_isdu_errors fatal_error)
  add_test(NAME startup_${scenario} COMMAND startup_recovery_tests ${scenario})
endforeach()
add_executable(isdu_regression_tests tests/isdu_regressions.cpp
    third_party/iolinki-master/tests/fake_iolink_device.c)
target_include_directories(isdu_regression_tests PRIVATE third_party/iolinki-master/tests)
target_link_libraries(isdu_regression_tests PRIVATE iolink_core)
if(MSVC)
  target_compile_options(isdu_regression_tests PRIVATE /W4 /utf-8)
endif()
foreach(scenario slow_response busy_timeout completion_idle completion_abort write_lengths segment_timeout)
  add_test(NAME isdu_${scenario} COMMAND isdu_regression_tests ${scenario})
endforeach()
add_executable(runtime_regression_tests tests/runtime_regressions.cpp
    third_party/iolinki-master/tests/fake_iolink_device.c)
target_include_directories(runtime_regression_tests PRIVATE third_party/iolinki-master/tests "${ARDUINOJSON_INCLUDE}")
target_link_libraries(runtime_regression_tests PRIVATE iolink_core)
if(MSVC)
  target_compile_options(runtime_regression_tests PRIVATE /W4 /utf-8)
endif()
foreach(scenario events_details events_legacy events_failures events_overflow events_isdu slow_max_cycle checksum_retries quality event_c_api)
  add_test(NAME runtime_${scenario} COMMAND runtime_regression_tests ${scenario})
endforeach()
